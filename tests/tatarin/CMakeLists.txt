cmake_minimum_required(VERSION 3.28)
project(TatarinChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
add_executable(tatarin-check main.cpp ../../launcher/tatarin/Repository.cpp)
target_include_directories(tatarin-check PRIVATE ../../launcher)
target_link_libraries(tatarin-check PRIVATE Qt6::Core Qt6::Network)
if(MSVC)
    target_compile_options(tatarin-check PRIVATE /utf-8 /W4 /WX)
endif()
